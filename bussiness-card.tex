\newcommand{\standhorizontaloffset}{-5.4mm}
\newcommand{\standverticaloffset}{-18.4mm}
\newcommand{\addOffsetsInMM}[2]{%
  \directlua{
    
    texio.write_nl("log", "========== OFFSET BEREKENING ==========")
    texio.write_nl("log", "Input 1: #1, Input 2: #2 ")
    local input1 = "#1"
    % texio.write_nl("log", "input1: " .. input1)
    local input2 = "#2"
    % texio.write_nl("log", "input2: " .. input2)
    local sp1 = tex.sp(input1)
    % texio.write_nl("log", "sp1: " .. sp1)
    local sp2 = tex.sp(input2)
    % texio.write_nl("log", "sp2: " .. sp2)
    local lsp = sp1 + sp2
    local mm_in_sp = tex.sp("1mm")
    % texio.write_nl("log", "mm_in_sp: " .. mm_in_sp)
    local sum_mm = lsp / mm_in_sp
    texio.write_nl("log", "Totaal in mm: " .. sum_mm)
    texio.write_nl("log", "=======================================")
    tex.print(sum_mm .. "mm")
  }%
}
